cmake_minimum_required(VERSION 3.21)
project(JkGfxModMaterialEditor LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.5 REQUIRED COMPONENTS Widgets OpenGL OpenGLWidgets)
add_executable(jkgm-material-editor WIN32 main.cpp)
target_link_libraries(jkgm-material-editor PRIVATE Qt6::Widgets Qt6::OpenGL Qt6::OpenGLWidgets)
if(MSVC)
  target_compile_options(jkgm-material-editor PRIVATE /W3 /EHsc /utf-8)
endif()
